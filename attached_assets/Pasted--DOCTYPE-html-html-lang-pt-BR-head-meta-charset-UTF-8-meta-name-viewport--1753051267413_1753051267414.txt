        <p class="portal-description">
            Encontre paz, respostas e renovação na quietude interior. Este portal convida você a desacelerar e a se reconectar com a sua essência através da respiração consciente.
        </p>

        <button id="startButton" class="action-button">Iniciar Prática de Silêncio</button>
        <p id="feedbackMessage" class="feedback-message"></p>
        <div id="breathingCounter" class="breathing-counter"></div>

        <button id="stopButton" class="action-button bg-red-500 hover:bg-red-600 hidden">Parar Prática</button>
    </div>

    <script>
        // Seleciona os elementos do DOM
        const startButton = document.getElementById('startButton');
        const stopButton = document.getElementById('stopButton');
        const feedbackMessage = document.getElementById('feedbackMessage');
        const breathingCounter = document.getElementById('breathingCounter');

        let breathingInterval; // Variável para armazenar o intervalo da respiração
        let count = 0; // Contador para a respiração
        let isBreathing = false; // Flag para controlar se a prática está ativa

        // Função para iniciar a prática de respiração
        function startBreathingPractice() {
            if (isBreathing) return; // Evita iniciar múltiplas vezes

            isBreathing = true;
            startButton.classList.add('hidden'); // Esconde o botão de iniciar
            stopButton.classList.remove('hidden'); // Mostra o botão de parar
            breathingCounter.classList.add('pulsing'); // Adiciona animação de pulsação

            let phase = 'Inspire'; // Fase inicial
            let duration = 4; // Duração de cada fase
            count = duration; // Inicia o contador com a duração

            feedbackMessage.textContent = `${phase}...`;
            breathingCounter.textContent = count;

            // Define o intervalo para a contagem da respiração
            breathingInterval = setInterval(() => {
                count--;
                breathingCounter.textContent = count;

                if (count === 0) {
                    // Alterna entre as fases de respiração
                    if (phase === 'Inspire') {
                        phase = 'Segure';
                        duration = 2;
                    } else if (phase === 'Segure') {
                        phase = 'Expire';
                        duration = 6;
                    } else if (phase === 'Expire') {
                        phase = 'Segure'; // Curta pausa antes de inspirar novamente
                        duration = 2;
                    } else if (phase === 'Segure') { // Após a pausa, volta a inspirar
                        phase = 'Inspire';
                        duration = 4;
                    }
                    count = duration;
                    feedbackMessage.textContent = `${phase}...`;
                    breathingCounter.textContent = count;
                }
            }, 1000); // Atualiza a cada segundo
        }

        // Função para parar a prática de respiração
        function stopBreathingPractice() {
            clearInterval(breathingInterval); // Limpa o intervalo
            isBreathing = false;
            startButton.classList.remove('hidden'); // Mostra o botão de iniciar
            stopButton.classList.add('hidden'); // Esconde o botão de parar
            feedbackMessage.textContent = "Prática finalizada. Sinta a quietude.";
            breathingCounter.textContent = ""; // Limpa o contador
            breathingCounter.classList.remove('pulsing'); // Remove animação
        }

        // Adiciona os event listeners aos botões
        startButton.addEventListener('click', startBreathingPractice);
        stopButton.addEventListener('click', stopBreathingPractice);

        // Mensagem inicial ao carregar a página
        window.onload = () => {
            feedbackMessage.textContent = "Clique para iniciar sua jornada de quietude.";
        };
    </script>
