const Button = ({ children, onClick, variant = "gold", size = "md", className = "" }) => {
  const base = "inline-flex items-center justify-center rounded-xl font-semibold transition active:scale-[0.99]";
  const sizes = {
    sm: "px-3 py-2 text-sm",
    md: "px-4 py-2.5 text-[0.95rem]",
    lg: "px-5 py-3 text-base",
  };
  const variants = {
    gold: `bg-[${BRAND.gold}] text-[#0b1220] hover:opacity-90`,
    ghost: `bg-transparent text-[${BRAND.gold}] border border-[${BRAND.gold}] hover:bg-[${BRAND.gold}] hover:text-[#0b1220]`,
    dark: `bg-[#0b1220] text-[${BRAND.gold}] border border-[${BRAND.gold}] hover:bg-[#0d1424]`,
  };
  return (
    <button onClick={onClick} className={[base, sizes[size], variants[variant], className].join(" ")}> 
      {children}
    </button>
  );
};

const ROUTES = ["login", "dashboard", "journal", "game", "reports", "community", "pipeline"] as const;

type Route = typeof ROUTES[number];

export default function App() {
  const [route, setRoute] = useState<Route>("login");
  const [user, setUser] = useState<{ name: string } | null>(null);

  const go = (r: Route) => setRoute(r);

  return (
    <div className="min-h-screen" style={{ background: BRAND.bg }}>
      {route !== "login" && (
        <TopBar onNavigate={go} />
      )}

      {route === "login" && <ScreenLogin onEnter={(u) => { setUser({ name: u }); go("dashboard"); }} onGuest={() => go("game")} />}
      {route === "dashboard" && <ScreenDashboard onNavigate={go} />}
      {route === "journal" && <ScreenJournal />}
      {route === "game" && <ScreenGame onNavigate={go} />}
      {route === "reports" && <ScreenReports />}
      {route === "community" && <ScreenCommunity />}
      {route === "pipeline" && <ScreenPipeline />}

      <Footer />
    </div>
  );
}
